var HOJA_USUARIOS = "Usuarios_Autorizados";
var HOJA_USUARIOS_LEGACY = "Usuarios";

/**
 * Función que verifica la existencia de la hoja Usuarios_Autorizados y determina el rol y permisos por módulo del usuario activo.
 * @param {string} [providedEmail] Correo opcional enviado desde la vista para verificación
 * @return {Object} Objeto JSON con success, registered, active, email, role y permissions por módulo
 */
function getCurrentUserRole(providedEmail) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      return {
        success: false,
        registered: false,
        active: false,
        email: "",
        role: "view",
        permissions: { Modulo_Inicio: "No", Modulo_Lotes: "No", Modulo_Gastos: "No", Modulo_Finanzas: "No" },
        error: "No se pudo obtener la hoja de cálculo activa."
      };
    }

    var sheet = ss.getSheetByName(HOJA_USUARIOS);
    var headersNeeded = ["ID", "Correo_Electronico", "Nombre", "Rol", "Estado", "Modulo_Inicio", "Modulo_Lotes", "Modulo_Gastos", "Modulo_Finanzas"];

    if (!sheet) {
      // Migración si existe la hoja Usuarios legacy
      var oldSheet = ss.getSheetByName(HOJA_USUARIOS_LEGACY);
      if (oldSheet) {
        try { oldSheet.setName(HOJA_USUARIOS); sheet = oldSheet; } catch(e) {}
      }
      if (!sheet) {
        sheet = ss.insertSheet(HOJA_USUARIOS);
        sheet.appendRow(headersNeeded);
      }
    }
    if (typeof asegurarColumnas === "function") {
      asegurarColumnas(sheet, headersNeeded);
    }

    // 1. Obtener correo del usuario activo de Google
    var activeEmail = "";
    try {
      if (typeof Session !== "undefined" && Session.getActiveUser) {
        activeEmail = Session.getActiveUser().getEmail();
      }
    } catch (e) { activeEmail = ""; }

    var effectiveEmail = "";
    try {
      if (typeof Session !== "undefined" && Session.getEffectiveUser) {
        effectiveEmail = Session.getEffectiveUser().getEmail();
      }
    } catch (e) { effectiveEmail = ""; }

    // Determinar correo a validar:
    // 1) Correo explícito enviado desde cliente/URL, o 2) Session activeEmail, o 3) Session effectiveEmail
    var userEmail = "";
    if (providedEmail && typeof providedEmail === "string" && providedEmail.trim()) {
      userEmail = providedEmail.trim().toLowerCase();
    } else if (activeEmail) {
      userEmail = activeEmail.trim().toLowerCase();
    } else if (effectiveEmail) {
      userEmail = effectiveEmail.trim().toLowerCase();
    }

    var lastRow = sheet.getLastRow();

    var defaultAdminPerms = { Modulo_Inicio: "Sí", Modulo_Lotes: "Sí", Modulo_Gastos: "Sí", Modulo_Finanzas: "Sí" };

    // Si la hoja Usuarios_Autorizados está vacía (solo encabezado), inicializar con la cuenta activa/proporcionada como primer Administrador
    if (lastRow < 2) {
      var initialAdmin = userEmail || activeEmail || effectiveEmail || "admin@fincasauces.com";
      initialAdmin = initialAdmin.trim().toLowerCase();
      sheet.appendRow(["USR-001", initialAdmin, initialAdmin.split('@')[0], "admin", "ACTIVO", "Sí", "Sí", "Sí", "Sí"]);
      return {
        success: true,
        registered: true,
        active: true,
        email: initialAdmin,
        role: "admin",
        permissions: defaultAdminPerms
      };
    }

    // Si no se proporcionó ningún correo ni se detectó en la sesión, solicitar ingreso
    if (!userEmail) {
      return {
        success: true,
        registered: false,
        active: false,
        email: "",
        role: "view",
        permissions: { Modulo_Inicio: "No", Modulo_Lotes: "No", Modulo_Gastos: "No", Modulo_Finanzas: "No" },
        error: "NO_EMAIL_DETECTED"
      };
    }

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    var emailColIdx = 1;
    var roleColIdx = 3;
    var statusColIdx = 4;
    var modInicioColIdx = -1;
    var modLotesColIdx = -1;
    var modGastosColIdx = -1;
    var modFinanzasColIdx = -1;

    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim().toLowerCase();
      if (h === "correo_electronico" || h === "email" || h === "correo") emailColIdx = c;
      if (h === "rol" || h === "role") roleColIdx = c;
      if (h === "estado" || h === "status") statusColIdx = c;
      if (h === "modulo_inicio") modInicioColIdx = c;
      if (h === "modulo_lotes") modLotesColIdx = c;
      if (h === "modulo_gastos") modGastosColIdx = c;
      if (h === "modulo_finanzas") modFinanzasColIdx = c;
    }

    var foundUser = null;
    for (var r = 0; r < values.length; r++) {
      var row = values[r];
      var rowEmail = row[emailColIdx] ? row[emailColIdx].toString().trim().toLowerCase() : "";
      if (rowEmail === userEmail) {
        var role = row[roleColIdx] ? row[roleColIdx].toString().trim().toLowerCase() : "view";
        var status = row[statusColIdx] ? row[statusColIdx].toString().trim().toUpperCase() : "INACTIVO";

        var isA = (role === "admin");
        var perms = {
          Modulo_Inicio: modInicioColIdx >= 0 ? (row[modInicioColIdx] || (isA ? "Sí" : "Sí")).toString().trim() : (isA ? "Sí" : "Sí"),
          Modulo_Lotes: modLotesColIdx >= 0 ? (row[modLotesColIdx] || (isA ? "Sí" : "Sí")).toString().trim() : (isA ? "Sí" : "Sí"),
          Modulo_Gastos: modGastosColIdx >= 0 ? (row[modGastosColIdx] || (isA ? "Sí" : "Sí")).toString().trim() : (isA ? "Sí" : "Sí"),
          Modulo_Finanzas: modFinanzasColIdx >= 0 ? (row[modFinanzasColIdx] || (isA ? "Sí" : "Sí")).toString().trim() : (isA ? "Sí" : "Sí")
        };

        foundUser = {
          email: rowEmail,
          role: isA ? "admin" : "view",
          status: status,
          permissions: perms
        };
        break;
      }
    }

    if (!foundUser) {
      return {
        success: true,
        registered: false,
        active: false,
        email: userEmail,
        role: "view",
        permissions: { Modulo_Inicio: "No", Modulo_Lotes: "No", Modulo_Gastos: "No", Modulo_Finanzas: "No" },
        error: "Acceso denegado: El correo " + userEmail + " no está en la lista blanca de Usuarios_Autorizados."
      };
    }

    var isActive = (foundUser.status === "ACTIVO" || foundUser.status === "ACTIVE");

    return {
      success: true,
      registered: true,
      active: isActive,
      email: foundUser.email,
      role: foundUser.role,
      permissions: foundUser.permissions
    };

  } catch (err) {
    return {
      success: false,
      registered: false,
      active: false,
      email: "",
      role: "view",
      permissions: { Modulo_Inicio: "No", Modulo_Lotes: "No", Modulo_Gastos: "No", Modulo_Finanzas: "No" },
      error: err.toString()
    };
  }
}

/**
 * Helper estricto de validación backend.
 * Retorna { autorizado: true, email: string, role: string, permissions: object } si la cuenta está registrada y activa en Usuarios_Autorizados.
 */
function verificarAccesoUsuario(emailOpcional) {
  var roleInfo = getCurrentUserRole(emailOpcional);
  if (roleInfo && roleInfo.registered && roleInfo.active && roleInfo.email) {
    return {
      autorizado: true,
      email: roleInfo.email,
      role: roleInfo.role,
      permissions: roleInfo.permissions
    };
  }
  return {
    autorizado: false,
    email: roleInfo ? roleInfo.email : "",
    role: "denied",
    permissions: { Modulo_Inicio: "No", Modulo_Lotes: "No", Modulo_Gastos: "No", Modulo_Finanzas: "No" },
    error: roleInfo && roleInfo.error ? roleInfo.error : "Acceso No Autorizado: La cuenta no se encuentra registrada o activa en el sistema."
  };
}

/**
 * Obtiene la lista completa de usuarios registrados en la hoja Usuarios.
 * @return {Object} Objeto JSON con success y la lista de usuarios.
 */
function obtenerListaUsuarios() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { success: false, error: "No se pudo obtener la hoja de cálculo activa." };

    var sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return { success: true, data: [] };

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: true, data: [] };

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    var result = [];
    for (var r = 0; r < values.length; r++) {
      var row = values[r];
      var userObj = {};
      var hasData = false;

      for (var c = 0; c < headers.length; c++) {
        var h = headers[c].toString().trim();
        var val = row[c] !== undefined && row[c] !== null ? row[c].toString().trim() : "";
        if (val) hasData = true;
        userObj[h] = val;
      }

      if (hasData) {
        var uEmail = (userObj.Correo_Electronico || userObj.Email || userObj.Correo || "").toLowerCase();
        result.push({
          id: userObj.ID || ("USR-" + String(r + 1).padStart(3, '0')),
          email: uEmail,
          nombre: userObj.Nombre || (uEmail ? uEmail.split('@')[0] : ""),
          rol: (userObj.Rol || userObj.Role || "view").toLowerCase() === "admin" ? "admin" : "view",
          estado: (userObj.Estado || userObj.Status || "ACTIVO").toUpperCase(),
          moduloInicio: userObj.Modulo_Inicio || "Sí",
          moduloLotes: userObj.Modulo_Lotes || "Sí",
          moduloGastos: userObj.Modulo_Gastos || "Sí",
          moduloFinanzas: userObj.Modulo_Finanzas || "Sí"
        });
      }
    }

    return { success: true, data: result };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

/**
 * Agrega o actualiza un usuario en la hoja Usuarios_Autorizados.
 * @param {Object} datos - Objeto con email, nombre, rol ("admin" o "view"), estado y permisos por módulo
 * @return {Object} Objeto JSON con el resultado de la operación.
 */
function guardarUsuario(datos) {
  try {
    var email = datos.email || datos.Email || datos.Correo_Electronico || "";
    var nombre = datos.nombre || datos.Nombre || (email ? email.split('@')[0] : "");
    var rol = (datos.rol || datos.Rol || "view").toLowerCase() === "admin" ? "admin" : "view";
    var estado = (datos.estado || datos.Estado || "ACTIVO").toUpperCase();

    var modInicio = (datos.Modulo_Inicio || datos.modulo_inicio || "Sí");
    var modLotes = (datos.Modulo_Lotes || datos.modulo_lotes || "Sí");
    var modGastos = (datos.Modulo_Gastos || datos.modulo_gastos || "Sí");
    var modFinanzas = (datos.Modulo_Finanzas || datos.modulo_finanzas || "Sí");

    email = email.trim().toLowerCase();
    if (!email) return { success: false, error: "El correo electrónico es obligatorio." };

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { success: false, error: "No se encontró la hoja de cálculo." };

    var sheet = ss.getSheetByName(HOJA_USUARIOS);
    var headersNeeded = ["ID", "Correo_Electronico", "Nombre", "Rol", "Estado", "Modulo_Inicio", "Modulo_Lotes", "Modulo_Gastos", "Modulo_Finanzas"];

    if (!sheet) {
      sheet = ss.insertSheet(HOJA_USUARIOS);
      sheet.appendRow(headersNeeded);
    }
    if (typeof asegurarColumnas === "function") {
      asegurarColumnas(sheet, headersNeeded);
    }

    var lastRow = sheet.getLastRow();
    var emailRowIdx = -1;

    if (lastRow >= 2) {
      var lastCol = sheet.getLastColumn();
      var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
      var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

      var emailColIdx = -1;
      for (var c = 0; c < headers.length; c++) {
        var h = headers[c].toString().trim().toLowerCase();
        if (h === "email" || h === "correo") {
          emailColIdx = c;
          break;
        }
      }
      if (emailColIdx === -1) emailColIdx = 1;

      for (var r = 0; r < values.length; r++) {
        var rowEmail = values[r][emailColIdx] ? values[r][emailColIdx].toString().trim().toLowerCase() : "";
        if (rowEmail === email) {
          emailRowIdx = r + 2;
          break;
        }
      }
    }

    if (emailRowIdx !== -1) {
      // Actualizar usuario existente
      sheet.getRange(emailRowIdx, 3).setValue(nombre);
      sheet.getRange(emailRowIdx, 4).setValue(rol);
      sheet.getRange(emailRowIdx, 5).setValue(estado);
      sheet.getRange(emailRowIdx, 6).setValue(modInicio);
      sheet.getRange(emailRowIdx, 7).setValue(modLotes);
      sheet.getRange(emailRowIdx, 8).setValue(modGastos);
      sheet.getRange(emailRowIdx, 9).setValue(modFinanzas);
      return { success: true, message: "Usuario " + email + " actualizado correctamente." };
    } else {
      // Insertar nuevo usuario
      var nextNum = lastRow < 2 ? 1 : (lastRow);
      var idUsr = "USR-" + String(nextNum).padStart(3, '0');
      sheet.appendRow([idUsr, email, nombre, rol, estado, modInicio, modLotes, modGastos, modFinanzas]);
      return { success: true, message: "Usuario " + email + " registrado con éxito." };
    }
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

/**
 * Elimina un usuario de la hoja Usuarios por correo electrónico.
 * @param {string} email - Correo del usuario a eliminar.
 * @return {Object} Objeto JSON con el resultado.
 */
function eliminarUsuario(email) {
  try {
    var targetEmail = String(email || "").trim().toLowerCase();
    if (!targetEmail) return { success: false, error: "Especifica un correo válido." };

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { success: false, error: "No existe la hoja de cálculo." };

    var sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return { success: false, error: "La hoja Usuarios no existe." };

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: false, error: "No hay usuarios registrados." };

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    var emailColIdx = -1;
    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim().toLowerCase();
      if (h === "email" || h === "correo") {
        emailColIdx = c;
        break;
      }
    }
    if (emailColIdx === -1) emailColIdx = 1;

    for (var r = 0; r < values.length; r++) {
      var rowEmail = values[r][emailColIdx] ? values[r][emailColIdx].toString().trim().toLowerCase() : "";
      if (rowEmail === targetEmail) {
        sheet.deleteRow(r + 2);
        return { success: true, message: "Usuario " + targetEmail + " eliminado correctamente." };
      }
    }

    return { success: false, error: "No se encontró el usuario con correo " + targetEmail };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}
